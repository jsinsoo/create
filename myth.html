Ⅳ. OOP 핵심 구조

원본 Tkinter 코드의 핵심 구조는 MythCharacter와 MythDictionary 클래스입니다. 인물 1명은 딕셔너리로 표현되고, 전체 인물은 리스트로 관리됩니다. 
class MythCharacter:
    def __init__(self, name, personality, role, category):
        self.data = {
            "인물": name,
            "성격": personality,
            "역할": role,
            "분류": category
        }

    def to_dict(self):
        return self.data


class MythDictionary:
    def __init__(self, raw_data):
        self.characters = []
        for item in raw_data:
            person = MythCharacter(
                item["인물"],
                item["성격"],
                item["역할"],
                item["분류"]
            )
            self.characters.append(person.to_dict())

    def find_by_name(self, name):
        for p in self.characters:
            if p["인물"] == name:
                return p
        return None

    def search(self, keyword):
        result = []
        for p in self.characters:
            text = p["인물"] + p["성격"] + p["역할"] + p["분류"]
            if keyword in text:
                result.append(p)
        return result
